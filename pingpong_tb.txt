library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity tb_ping_pong is
end tb_ping_pong;

architecture Behavioral of tb_ping_pong is
    component ping_pong
        Port ( 
            i_clk   : in  STD_LOGIC;
            i_rst   : in  STD_LOGIC;
            i_btnL  : in  STD_LOGIC;
            i_btnR  : in  STD_LOGIC;
            o_led   : out STD_LOGIC_VECTOR (7 downto 0)
        );
    end component;

    signal i_clk  : STD_LOGIC := '0';
    signal i_rst  : STD_LOGIC := '1';
    signal i_btnL : STD_LOGIC := '0';
    signal i_btnR : STD_LOGIC := '0';
    signal o_led  : STD_LOGIC_VECTOR (7 downto 0);

    constant CLK_PERIOD : time := 20 ns;
begin
    uut: ping_pong port map (
        i_clk  => i_clk,
        i_rst  => i_rst,
        i_btnL => i_btnL,
        i_btnR => i_btnR,
        o_led  => o_led
    );

    -- 產生時脈 50MHz
    clk_process : process
    begin
        i_clk <= '0';
        wait for CLK_PERIOD/2;
        i_clk <= '1';
        wait for CLK_PERIOD/2;
    end process;

    -- 測試劇本
    stim_proc: process
    begin
        -- 1. 初始化與重置
        i_rst <= '1';
        i_btnL <= '0';
        i_btnR <= '0';
        wait for 100 ns;
        
        wait until falling_edge(i_clk);
        i_rst <= '0';
        wait for CLK_PERIOD * 2;

        -- 2. 左方發球
        wait until falling_edge(i_clk);
        i_btnL <= '1';
        wait until falling_edge(i_clk);
        i_btnL <= '0';
        
        wait for CLK_PERIOD * 8; 

        -- 3. 右方精準回擊 (在最右邊緣按下)
        wait until falling_edge(i_clk);
        i_btnR <= '1';
        wait until falling_edge(i_clk);
        i_btnR <= '0';

        -- 球往左飛回 (需要 7 拍才會到最左邊 10000000)
        wait for CLK_PERIOD * 7;

        -- 4. 左方漏接 (球到了左邊，故意不按按鍵，讓時脈繼續走)
        wait for CLK_PERIOD * 5;
        
        -- 5. 左方重新發球 (此時畫面應該已經顯示右方得一分：00000001)
        wait until falling_edge(i_clk);
        i_btnL <= '1';
        wait until falling_edge(i_clk);
        i_btnL <= '0';

        wait;
    end process;
end Behavioral;
